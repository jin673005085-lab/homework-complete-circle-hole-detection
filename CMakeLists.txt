cmake_minimum_required(VERSION 3.15)
project(CircleHoles LANGUAGES CXX)
find_package(OpenCV REQUIRED)
add_executable(circle_holes main.cpp)
target_compile_features(circle_holes PRIVATE cxx_std_17)
target_include_directories(circle_holes PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(circle_holes PRIVATE ${OpenCV_LIBS})
if(WIN32)
    target_link_libraries(circle_holes PRIVATE shell32)
endif()
